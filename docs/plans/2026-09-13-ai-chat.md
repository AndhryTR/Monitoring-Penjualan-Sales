### Task 2: aiSettings (setelan + storage device)

**Files:**
- Create: `src/utils/aiSettings.js`
- Test: `smoke-ai-settings.mjs` (sementara, hapus sesudah)

**Step 1: Write minimal implementation**

```js
const KEY = "smapp:ai_settings:v1";
const DEFAULTS = { mode: "direct", apiType: "openai-compat", baseURL: "", model: "", key: "", backendURL: "", proxyKey: "" };
export function loadAiSettings() {
  try {
    const raw = window.localStorage.getItem(KEY);
    return raw ? { ...DEFAULTS, ...JSON.parse(raw) } : { ...DEFAULTS };
  } catch { return { ...DEFAULTS }; }
}
export function saveAiSettings(s) {
  window.localStorage.setItem(KEY, JSON.stringify({ ...DEFAULTS, ...s }));
}
export function clearAiKey() {
  const s = loadAiSettings(); s.key = "";
  saveAiSettings(s);
}
```

**Step 2: Smoke** — load default + roundtrip save/load di node (mock localStorage minimal).
